import numpy as np 

np.random.seed(42)

# defining the backprop for Sigmoid Function
def back_sigmoid(x):
    return x*(1-x)

# defining the Sigmoid Function
def sigmoid(x):
    return 1 / (1 + np.exp(-x))

def forward_pass(X, weights_input_hidden, weights_hidden_output):
    # Hidden layer linear combination: (N, 1) @ (1, 3) -> (N, 3)
    a_hidden = X @ weights_input_hidden # a_2, a_3, a_4

    # Hidden layer activation (Sigmoid): z_2, z_3, z_4
    hiddenLayer_activations = sigmoid(a_hidden)

    # Output layer linear combination: (N, 3) @ (3, 1) -> (N, 1)
    a_output = hiddenLayer_activations @ weights_hidden_output # a_1

    # Output layer activation (Sigmoid): z_1
    output = sigmoid(a_output) # z_1
    
    return output, hiddenLayer_activations # z_1, [z_2, z_3, z_4]


def backward_pass(X, y, output, weights_hidden_output, weights_input_hidden, hiddenLayer_activations):
    # calculating rate of change of error w.r.t weight between hidden and output layer

    # Output layer error (δ_1)
    delta_1 = (output - y) * back_sigmoid(output) # shape: (N, 1)

    # Gradient for output layer's weights
    # hiddenLayer_activation.T: (3, N), delta_1: (N, 1) -> error_wrt_weights_hidden_output: (3, 1) 
    error_wrt_weights_hidden_output = hiddenLayer_activations.T @ delta_1 # δ_1 * z_2, z_3, z_4. z_i is like how much the neuron contributed to the output of the model. 

    # Hidden layer errors (δ_2, δ_3, δ_4)
    # delta_1 @ weights_hidden_output.T: (N, 1) @ (1, 3) → (N, 3)
    delta_hidden = back_sigmoid(hiddenLayer_activations) * (delta_1 @ weights_hidden_output.T)

    # Gradient for hidden layer's weights
    # X.T: (1, N), delta_hidden: (N, 3) -> error_wrt_weights_input_hidden: (1, 3)
    error_wrt_weights_input_hidden = X.T @ delta_hidden

    return error_wrt_weights_hidden_output, error_wrt_weights_input_hidden


def train(X_train, y_train):

    # defining the model architecture
    inputLayer_neurons = 1  # number of neurons at input
    hiddenLayer_neurons = 3  # number of hidden layers neurons
    outputLayer_neurons = 1  # number of neurons at output layer

    # initializing weight
    weights_input_hidden = np.random.uniform(size=(inputLayer_neurons, hiddenLayer_neurons))
    weights_hidden_output = np.random.uniform(
        size=(hiddenLayer_neurons, outputLayer_neurons)
    )
    # defining the parameters
    lr = 0.1
    epochs = 1000

    losses =  []

    for ep in range(epochs):
        epoch_losses = []

        for i in range(X_train.shape[0]):
            x = X_train[i].reshape(1, 1)
            y = y_train[i].reshape(1, 1)
            
            output_, hiddenLayer_activations = forward_pass(x, weights_input_hidden, weights_hidden_output)

            loss = np.square(y - output_) / 2
            epoch_losses.append(loss.item())

            ## Backward Propagation
            error_wrt_weights_hidden_output, error_wrt_weights_input_hidden = backward_pass(x, y, 
                                                                                            output_, 
                                                                                            weights_hidden_output, 
                                                                                            weights_input_hidden, 
                                                                                            hiddenLayer_activations)
            # updating the weights
            weights_hidden_output -= lr * error_wrt_weights_hidden_output
            weights_input_hidden -= lr * error_wrt_weights_input_hidden

        avg_loss = np.mean(epoch_losses)
        losses.append(avg_loss)
        if ep % 100 == 0:
            print(f"Epoch {ep}: Loss = {avg_loss:.5f}")

    import matplotlib.pyplot as plt
    plt.plot(losses)
    plt.show()

    final_pred, _ = forward_pass(X_train, weights_input_hidden, weights_hidden_output)

    plt.plot(final_pred, c='r')
    plt.plot(y_train, c='b')
    plt.show()

    print("Weights of the network are: ")
    print("w13, w12, w14", weights_hidden_output.T)
    print("w35, w25, w45", weights_input_hidden)
